class Solution:    def connect(self, root):        cur = root.        while cur:            dummy = Node(0)            nxt = dummy.            while cur:                if cur.left:                    nxt.next = cur.left                    nxt = nxt.next.                if cur.right:                    nxt.next = cur.right                    nxt = nxt.next.                cur = cur.next.            cur = dummy.next.        return root
